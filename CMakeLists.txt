cmake_minimum_required(VERSION 3.13)

if(NOT PICO_SDK_PATH)
    message(FATAL_ERROR "PICO_SDK_PATH is not set; build with `make` instead.")
endif()
set(PICO_BOARD pico2_w CACHE STRING "Board type")

include(${PICO_SDK_PATH}/pico_sdk_init.cmake)

project(pico2w_wigle C CXX ASM)
set(CMAKE_C_STANDARD 11)

pico_sdk_init()

set(SOURCES
    src/clock.c
    src/main.c
    src/net.c
    src/scan.c
    src/upload.c
)
add_executable(pico2w-wigle ${SOURCES})

# Only for our own files; SDK library sources are compiled into the same
# target.
set_source_files_properties(${SOURCES} PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra")

# config.h is generated by scripts/gen-config.sh.
target_include_directories(pico2w-wigle PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}/src
    ${CMAKE_BINARY_DIR}/generated
)

target_compile_definitions(pico2w-wigle PRIVATE
    CYW43_HOST_NAME=\"pico2w-wigle\"
    # Give a serial terminal time to attach so the boot messages are seen.
    PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=3000
)

target_link_libraries(pico2w-wigle
    hardware_watchdog
    pico_cyw43_arch_lwip_poll
    pico_lwip_sntp
    pico_mbedtls
    pico_stdlib
)

pico_enable_stdio_usb(pico2w-wigle 1)
pico_enable_stdio_uart(pico2w-wigle 0)
pico_add_extra_outputs(pico2w-wigle)
